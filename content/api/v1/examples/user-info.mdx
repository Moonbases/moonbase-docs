<Callout type="info" title="Legacy API">
  Send the public app key (`PUBLIC_API_KEY` from the app's `.env`) in `Authorization` and the JWT from [sign-in](/docs/api/v1/examples/authentication) in `Token`. Replace `https://your-moonbase-host` with your API URL.
</Callout>

Every example returns `status_data.user` and fails with `status_message` otherwise (`INVALID_TOKEN` means the JWT expired: sign in again). See [User profile](/docs/api/v1/user-info) for the fields.

```typescript
type V1Response<T> = {
  status_message: string;
  status_overview: "success" | "failed";
  status_code: number;
  status_data?: T;
};

type OwnedProduct = { product: string; productName: string; expiration_date: string; seats?: number };
type V1User = { _id: string; username: string; email: string; flags: number; products: OwnedProduct[]; hwid: string[]; ip: string[] };

export async function getUserInfo(appKey: string, jwt: string): Promise<V1User> {
  const res = await fetch("https://your-moonbase-host/api/user/info", {
    headers: { Authorization: appKey, Token: jwt, Accept: "application/json" },
  });
  const body = (await res.json()) as V1Response<{ user: V1User }>;
  if (body.status_overview !== "success" || !body.status_data) {
    throw new Error(`${body.status_code} ${body.status_message}`);
  }
  return body.status_data.user;
}
```

```java
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;

public class MoonbaseUserInfo {
    private static final HttpClient CLIENT = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(10)).build();
    private static final ObjectMapper JSON = new ObjectMapper();

    public static JsonNode getUserInfo(String appKey, String jwt) throws Exception {
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://your-moonbase-host/api/user/info"))
            .header("Authorization", appKey)
            .header("Token", jwt)
            .timeout(Duration.ofSeconds(30))
            .GET()
            .build();

        JsonNode body = JSON.readTree(CLIENT.send(request, HttpResponse.BodyHandlers.ofString()).body());
        if (!"success".equals(body.path("status_overview").asText())) {
            throw new IllegalStateException(body.path("status_message").asText());
        }
        return body.path("status_data").path("user");
    }
}
```

```cpp
// libcurl + nlohmann/json
#include <curl/curl.h>
#include <nlohmann/json.hpp>
#include <stdexcept>
#include <string>

static size_t collect(char* data, size_t size, size_t count, void* out) {
  static_cast<std::string*>(out)->append(data, size * count);
  return size * count;
}

nlohmann::json getUserInfo(const std::string& appKey, const std::string& jwt) {
  std::string response;
  CURL* curl = curl_easy_init();
  if (!curl) throw std::runtime_error("curl_easy_init failed");
  curl_slist* headers = nullptr;
  headers = curl_slist_append(headers, ("Authorization: " + appKey).c_str());
  headers = curl_slist_append(headers, ("Token: " + jwt).c_str());
  curl_easy_setopt(curl, CURLOPT_URL, "https://your-moonbase-host/api/user/info");
  curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
  curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, collect);
  curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response);
  curl_easy_setopt(curl, CURLOPT_TIMEOUT, 30L);
  const CURLcode result = curl_easy_perform(curl);
  curl_slist_free_all(headers);
  curl_easy_cleanup(curl);
  if (result != CURLE_OK) throw std::runtime_error(curl_easy_strerror(result));

  const auto body = nlohmann::json::parse(response);
  if (body.value("status_overview", "") != "success") throw std::runtime_error(body.value("status_message", "REQUEST_FAILED"));
  return body["status_data"]["user"];
}
```

```go
package moonbase

import (
	"encoding/json"
	"fmt"
	"net/http"
	"time"
)

type OwnedProduct struct {
	Product        string `json:"product"`
	ProductName    string `json:"productName"`
	ExpirationDate string `json:"expiration_date"`
}

type User struct {
	ID       string         `json:"_id"`
	Username string         `json:"username"`
	Email    string         `json:"email"`
	Products []OwnedProduct `json:"products"`
	HWID     []string       `json:"hwid"`
}

func GetUserInfo(appKey, jwt string) (*User, error) {
	req, _ := http.NewRequest(http.MethodGet, "https://your-moonbase-host/api/user/info", nil)
	req.Header.Set("Authorization", appKey)
	req.Header.Set("Token", jwt)

	resp, err := (&http.Client{Timeout: 30 * time.Second}).Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	var body struct {
		StatusMessage  string `json:"status_message"`
		StatusOverview string `json:"status_overview"`
		StatusCode     int    `json:"status_code"`
		StatusData     struct {
			User User `json:"user"`
		} `json:"status_data"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&body); err != nil {
		return nil, err
	}
	if body.StatusOverview != "success" {
		return nil, fmt.Errorf("%d %s", body.StatusCode, body.StatusMessage)
	}
	return &body.StatusData.User, nil
}
```

```swift
import Foundation

struct OwnedProduct: Decodable {
    let product: String
    let productName: String
    let expiration_date: String
}

struct V1User: Decodable {
    let _id: String
    let username: String
    let email: String
    let products: [OwnedProduct]
}

struct UserInfoResponse: Decodable {
    struct Payload: Decodable { let user: V1User }
    let status_message: String
    let status_overview: String
    let status_data: Payload?
}

enum MoonbaseError: Error { case failed(String) }

func getUserInfo(appKey: String, jwt: String) async throws -> V1User {
    var request = URLRequest(url: URL(string: "https://your-moonbase-host/api/user/info")!)
    request.setValue(appKey, forHTTPHeaderField: "Authorization")
    request.setValue(jwt, forHTTPHeaderField: "Token")
    request.timeoutInterval = 30

    let (data, _) = try await URLSession.shared.data(for: request)
    let body = try JSONDecoder().decode(UserInfoResponse.self, from: data)
    guard body.status_overview == "success", let result = body.status_data else {
        throw MoonbaseError.failed(body.status_message)
    }
    return result.user
}
```

```lua
-- LuaSocket + LuaSec (HTTPS) + dkjson
local https = require("ssl.https")
local ltn12 = require("ltn12")
local json = require("dkjson")

local function get_user_info(app_key, jwt)
  local chunks = {}
  local ok, code = https.request({
    url = "https://your-moonbase-host/api/user/info",
    method = "GET",
    headers = { ["Authorization"] = app_key, ["Token"] = jwt },
    sink = ltn12.sink.table(chunks),
  })
  if not ok then error("request failed: " .. tostring(code)) end

  local body = json.decode(table.concat(chunks))
  if not body or body.status_overview ~= "success" then
    error(body and body.status_message or ("HTTP " .. tostring(code)))
  end
  return body.status_data.user
end

local user = get_user_info("<PUBLIC_API_KEY>", "<jwt_token>")
for _, entry in ipairs(user.products) do print(entry.productName, entry.expiration_date) end
```
